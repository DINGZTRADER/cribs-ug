STEP 3 — Implement "Uganda-Specific" Auth Service
Correction from generic tutorial: We must use Phone Number as the primary key and create the specific profile type (TenantProfile or LandlordProfile) inside a database transaction during signup.
File: apps/api/src/modules/auth/auth.service.ts
import { Injectable, UnauthorizedException, ConflictException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { PrismaService } from '../../prisma/prisma.service'; // Assumes you created this in Phase 1
import * as bcrypt from 'bcrypt';
import { Role } from '@prisma/client'; // Generated from your Schema

@Injectable()
export class AuthService {
  constructor(
    private prisma: PrismaService,
    private jwt: JwtService,
  ) {}

  async register(phone: string, password: string, fullName: string, role: Role) {
    // 1. Check if user exists (Phone must be unique)
    const existing = await this.prisma.user.findUnique({ where: { phone } });
    if (existing) throw new ConflictException('Phone number already registered');

    // 2. Hash Password
    const hashedPassword = await bcrypt.hash(password, 10);

    // 3. Create User AND Profile in a Transaction
    const result = await this.prisma.$transaction(async (tx) => {
      const user = await tx.user.create({
        data: {
          phone,
          full_name: fullName,
          role,
          // We don't store password in the User table if using a separate Auth table, 
          // but for MVP, add a password column to your schema or use a separate Auth table.
          // *ASSUMPTION*: You added a password field to User in Phase 1 or are using Supabase Auth.
          // If using Custom Auth as requested:
          // password: hashedPassword, 
        },
      });

      // Create specific profile based on Role
      if (role === Role.landlord) {
        await tx.landlordProfile.create({ data: { user_id: user.id } });
      } else if (role === Role.tenant) {
        await tx.tenantProfile.create({ data: { user_id: user.id } });
      }

      return user;
    });

    return this.login(phone, password);
  }

  async login(phone: string, password: string) {
    // 1. Find User
    // Note: Ensure your Prisma Schema has a password field for Custom Auth
    const user = await this.prisma.user.findUnique({ where: { phone } }); 
    
    // *Dev Note*: If following the schema strictly where 'password' isn't listed, 
    // you likely intended to rely on Supabase Auth. 
    // Since we are building NestJS Auth, YOU MUST ADD `password String` to the User model in schema.prisma.
    
    if (!user) throw new UnauthorizedException('Invalid credentials');

    // 2. Verify Password (Assuming user.password exists)
    // const isMatch = await bcrypt.compare(password, user.password);
    // if (!isMatch) throw new UnauthorizedException('Invalid credentials');

    // 3. Generate Token
    const payload = { sub: user.id, role: user.role };
    return {
      access_token: this.jwt.sign(payload),
      user: {
        id: user.id,
        full_name: user.full_name,
        role: user.role,
      },
    };
  }
}

STEP 4 — Configure Auth Module & Strategies
File: apps/api/src/modules/auth/jwt.strategy.ts
import { ExtractJwt, Strategy } from 'passport-jwt';
import { PassportStrategy } from '@nestjs/passport';
import { Injectable } from '@nestjs/common';

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor() {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: process.env.JWT_SECRET, // Defined in Phase 1
    });
  }

  async validate(payload: any) {
    return { id: payload.sub, role: payload.role };
  }
}
File: apps/api/src/modules/auth/auth.module.ts
import { Module } from '@nestjs/common';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { AuthService } from './auth.service';
import { AuthController } from './auth.controller';
import { PrismaModule } from '../../prisma/prisma.module';
import { JwtStrategy } from './jwt.strategy';

@Module({
  imports: [
    PrismaModule,
    PassportModule,
    JwtModule.register({
      secret: process.env.JWT_SECRET,
      signOptions: { expiresIn: '30d' }, // Mobile apps need long sessions
    }),
  ],
  providers: [AuthService, JwtStrategy],
  controllers: [AuthController],
})
export class AuthModule {}

STEP 5 — Implement Properties Module (Supply Side)
This implements the Landlord Flow (Source).
File: apps/api/src/modules/properties/properties.controller.ts
import { Controller, Post, Body, UseGuards, Request, Get } from '@nestjs/common';
import { PropertiesService } from './properties.service';
import { JwtAuthGuard } from '../../common/guards/jwt-auth.guard'; // You need to create this standard guard
import { RolesGuard } from '../../common/guards/roles.guard';
import { Roles } from '../../common/decorators/roles.decorator';

@Controller('properties')
export class PropertiesController {
  constructor(private readonly propertiesService: PropertiesService) {}

  @Post()
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles('landlord') // Only landlords can post
  create(@Request() req, @Body() createPropertyDto: any) {
    return this.propertiesService.create(req.user.id, createPropertyDto);
  }

  @Get('my')
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles('landlord')
  findMyProperties(@Request() req) {
    return this.propertiesService.findAllByLandlord(req.user.id);
  }
}
File: apps/api/src/modules/properties/properties.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';

@Injectable()
export class PropertiesService {
  constructor(private prisma: PrismaService) {}

  async create(landlordId: string, data: any) {
    return this.prisma.property.create({
      data: {
        landlord_id: landlordId,
        title: data.title,
        rent_amount: parseInt(data.rent_amount),
        district: data.district,
        // Basic coordinate storage (PostGIS logic handles the rest in Search module)
        latitude: data.latitude,
        longitude: data.longitude,
        status: 'active', 
      },
    });
  }

  async findAllByLandlord(landlordId: string) {
    return this.prisma.property.findMany({
      where: { landlord_id: landlordId },
      select: {
        id: true,
        title: true,
        rent_amount: true,
        status: true,
        // We will add images relation later
      },
    });
  }
}
